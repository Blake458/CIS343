==================================================
           SYNTACTICAL GRAMMAR DESIGN          
               by Blake Collins
==================================================

program        -> statement* EOF ;

declaration    -> varDecl
                | statement

statement      -> printStmt
                | ifStmt
                | block
                | expressionStmt ;

ifStmt         -> "if" "(" expression ")" statement
                ("else" statement )? ;

printStmt      -> "print" expression ";" ;

varDecl        -> "new" IDENTIFIER ("=" expression)? ";" ;

block          -> "{" statement* "}" ;

expressionStmt -> expression ";" ;

expression     -> assignment ;

assignment     -> IDENTIFIER "=" assignment
                | equality ;

equality       -> comparison ( ( "==" | "!=" ) comparison )* ;

comparison     -> term ( ( ">" | ">=" | "<" | "<=" ) term )* ;

term           -> factor ( ( "+" | "-" ) factor )* ;

factor         -> unary ( ( "*" | "/" ) unary )* ;

unary          -> ( "-" | "!" ) unary
                | primary ;

primary        -> NUMBER
                | STRING
                | "True"
                | "False"
                | "null"
                | IDENTIFIER
                | "(" expression ")" ;
